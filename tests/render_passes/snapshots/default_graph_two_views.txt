1. Light Update
    create Lights HostWrite
2. Instance Update
    create StaticInstances HostWrite
    create DynamicInstances HostWrite
    create DrawBatches HostWrite
3. Edit/Scene Update
    create Edit/CameraData HostWrite
4. Edit/Culling
    use    Edit/CameraData ShaderRead
    use    StaticInstances ShaderRead
    use    DynamicInstances ShaderRead
    use    DrawBatches ShaderRead
    create Edit/VisibleInstances StorageWrite
    create Edit/IndirectDrawCommands StorageWrite
    barrier Edit/VisibleInstances: last frame's ShaderRead -> StorageWrite
    barrier Edit/IndirectDrawCommands: last frame's IndirectRead -> StorageWrite
5. Edit/Draw Compaction

    use    Edit/VisibleInstances StorageWrite
    use    Edit/IndirectDrawCommands StorageWrite
