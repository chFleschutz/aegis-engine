1. Main/Scene Update
    create Main/CameraData HostWrite
2. Main/Geometry
    use    Main/CameraData ShaderRead
    create Main/Position ColorAttachment
    create Main/Normal ColorAttachment
    create Main/Albedo ColorAttachment
    create Main/ARM ColorAttachment
    create Main/Emissive ColorAttachment
    create Main/Depth DepthAttachment
    barrier Main/Position: undefined -> ColorAttachment
    barrier Main/Normal: undefined -> ColorAttachment
    barrier Main/Albedo: undefined -> ColorAttachment
    barrier Main/ARM: undefined -> ColorAttachment
    barrier Main/Emissive: undefined -> ColorAttachment
    barrier Main/Depth: undefined -> DepthAttachment
3. Main/Lighting
    use    Main/Position StorageRead
    use    Main/Normal StorageRead
    use    Main/Albedo StorageRead
    use    Main/ARM StorageRead
    use    Main/Emissive StorageRead
    create Main/SceneColor StorageWrite
    create Main/LightingUniforms HostWrite
    barrier Main/Position: ColorAttachment -> StorageRead
    barrier Main/Normal: ColorAttachment -> StorageRead
    barrier Main/Albedo: ColorAttachment -> StorageRead
    barrier Main/ARM: ColorAttachment -> StorageRead
    barrier Main/Emissive: ColorAttachment -> StorageRead
    barrier Main/SceneColor: undefined -> StorageWrite
4. Main/Sky Box
    use    Main/Depth DepthRead
    use    Main/SceneColor ColorAttachment
    barrier Main/Depth: DepthAttachment -> DepthRead
    barrier Main/SceneColor: StorageWrite -> ColorAttachment
5. Main/Transparent
    use    Main/Depth DepthRead
    use    Main/CameraData ShaderRead
    use    Main/SceneColor ColorAttachment
    barrier Main/SceneColor: ColorAttachment -> ColorAttachment
6. Main/Bloom
    use    Main/SceneColor StorageRead
    create Main/Bloom StorageWrite
    barrier Main/SceneColor: ColorAttachment -> StorageRead
    barrier Main/Bloom: undefined -> StorageWrite
7. Main/Post Processing
    use    Main/SceneColor StorageRead
    use    Main/Bloom StorageRead
    create Main/Final StorageWrite
    barrier Main/Bloom: StorageWrite -> StorageRead
    barrier Main/Final: undefined -> StorageWrite
8. UI
    use    Main/Final SampledFragment
    barrier Main/Final: StorageWrite -> SampledFragment
resources
    Main/CameraData: buffer, host written
    Main/Position: texture RGBA16_SFLOAT
    Main/Normal: texture RGBA16_SFLOAT
    Main/Albedo: texture RGBA8_UNORM
    Main/ARM: texture RGBA8_UNORM
    Main/Emissive: texture RGBA8_UNORM
    Main/Depth: texture D32_SFLOAT
    Main/SceneColor: texture RGBA16_SFLOAT
    Main/LightingUniforms: buffer, host written
    Main/Bloom: texture RGBA16_SFLOAT
    Main/Final: texture RGBA16_SFLOAT
