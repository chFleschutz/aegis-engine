1. Light Update
    create Lights HostWrite
2. Instance Update
    create StaticInstances HostWrite
    create DynamicInstances HostWrite
    create DrawBatches HostWrite
3. Main/Scene Update
    create Main/CameraData HostWrite
4. Main/Culling
    use    Main/CameraData ShaderRead
    use    StaticInstances ShaderRead
    use    DynamicInstances ShaderRead
    use    DrawBatches ShaderRead
    create Main/VisibleInstances StorageWrite
    create Main/IndirectDrawCommands StorageWrite
    barrier Main/VisibleInstances: last frame's ShaderRead -> StorageWrite
    barrier Main/IndirectDrawCommands: last frame's IndirectRead -> StorageWrite
5. Main/Draw Compaction
    use    DrawBatches ShaderRead
    use    Main/VisibleInstances StorageWrite
    use    Main/IndirectDrawCommands StorageWrite
    create Main/IndirectDrawCounts StorageWrite
    barrier Main/VisibleInstances: StorageWrite -> StorageWrite
    barrier Main/IndirectDrawCommands: StorageWrite -> StorageWrite
    barrier Main/IndirectDrawCounts: last frame's IndirectRead -> StorageWrite
6. Main/GPU Driven Geometry
    use    Main/CameraData ShaderRead
    use    StaticInstances ShaderRead
    use    DynamicInstances ShaderRead
    use    Main/VisibleInstances ShaderRead
    use    Main/IndirectDrawCommands IndirectRead
    use    Main/IndirectDrawCounts IndirectRead
    create Main/Position ColorAttachment
    create Main/Normal ColorAttachment
    create Main/Albedo ColorAttachment
    create Main/ARM ColorAttachment
    create Main/Emissive ColorAttachment
    create Main/MotionVectors ColorAttachment
    create Main/Depth DepthAttachment
    barrier Main/VisibleInstances: StorageWrite -> ShaderRead
    barrier Main/IndirectDrawCommands: StorageWrite -> IndirectRead
    barrier Main/IndirectDrawCounts: StorageWrite -> IndirectRead
    barrier Main/Position: undefined -> ColorAttachment
    barrier Main/Normal: undefined -> ColorAttachment
    barrier Main/Albedo: undefined -> ColorAttachment
    barrier Main/ARM: undefined -> ColorAttachment
    barrier Main/Emissive: undefined -> ColorAttachment
    barrier Main/MotionVectors: undefined -> ColorAttachment
    barrier Main/Depth: undefined -> DepthAttachment
7. Main/Lighting
    use    Main/Position StorageRead
    use    Main/Normal StorageRead
    use    Main/Albedo StorageRead
    use    Main/ARM StorageRead
    use    Main/Emissive StorageRead
    use    Lights ShaderRead
    create Main/SceneColor StorageWrite
    create Main/LightingUniforms HostWrite
    barrier Main/Position: ColorAttachment -> StorageRead
    barrier Main/Normal: ColorAttachment -> StorageRead
    barrier Main/Albedo: ColorAttachment -> StorageRead
    barrier Main/ARM: ColorAttachment -> StorageRead
    barrier Main/Emissive: ColorAttachment -> StorageRead
    barrier Main/SceneColor: undefined -> StorageWrite
8. Main/Sky Box
    use    Main/Depth DepthRead
    use    Main/SceneColor ColorAttachment
    barrier Main/Depth: DepthAttachment -> DepthRead
    barrier Main/SceneColor: StorageWrite -> ColorAttachment
9. Main/Forward
    use    Main/Depth DepthRead
    use    Main/CameraData ShaderRead
    use    Main/LightingUniforms ShaderRead
    use    Lights ShaderRead
    use    Main/SceneColor ColorAttachment
    barrier Main/SceneColor: ColorAttachment -> ColorAttachment
10. Main/Transparent
    use    Main/Depth DepthRead
    use    Main/CameraData ShaderRead
    use    Main/SceneColor ColorAttachment
    barrier Main/SceneColor: ColorAttachment -> ColorAttachment
11. Main/Temporal AA
    use    Main/SceneColor StorageRead
    use    Main/Depth SampledCompute
    use    Main/MotionVectors SampledCompute
    use    Main/TemporalColor SampledCompute (previous frame)
    create Main/TemporalColor StorageWrite
    barrier Main/SceneColor: ColorAttachment -> StorageRead
    barrier Main/MotionVectors: ColorAttachment -> SampledCompute
    barrier Main/TemporalColor (previous frame): last frame's StorageRead -> SampledCompute
    barrier Main/TemporalColor: last frame's SampledCompute -> StorageWrite
12. Main/Bloom
    use    Main/TemporalColor StorageRead
    create Main/Bloom StorageWrite
    barrier Main/TemporalColor: StorageWrite -> StorageRead
    barrier Main/Bloom: undefined -> StorageWrite
13. Main/Post Processing
    use    Main/TemporalColor StorageRead
    use    Main/Bloom StorageRead
    create Main/Final StorageWrite
    barrier Main/Bloom: StorageWrite -> StorageRead
    barrier Main/Final: undefined -> StorageWrite
14. UI
    use    Main/Final ColorAttachment
    barrier Main/Final: StorageWrite -> ColorAttachment
15. Present
    use    Main/Final TransferSrc
    import Target TransferDst
    barrier Main/Final: ColorAttachment -> TransferSrc
    barrier Target: ColorAttachment -> TransferDst
end of frame
    barrier Target: TransferDst -> ColorAttachment
resources

    DynamicInstances: buffer, host written
    DrawBatches: buffer, host written
    Main/CameraData: buffer, host written
    Main/VisibleInstances: buffer
    Main/IndirectDrawCommands: buffer
    Main/IndirectDrawCounts: buffer
    Main/Position: texture RGBA16_SFLOAT
    Main/Normal: texture RGBA16_SFLOAT
    Main/Albedo: texture RGBA8_UNORM
    Main/ARM: texture RGBA8_UNORM
    Main/Emissive: texture RGBA8_UNORM
    Main/MotionVectors: texture RG16_SFLOAT
    Main/Depth: texture D32_SFLOAT
    Main/SceneColor: texture RGBA16_SFLOAT
    Main/LightingUniforms: buffer, host written
    Main/TemporalColor: texture RGBA16_SFLOAT, history pair
    Main/Bloom: texture RGBA16_SFLOAT
    Main/Final: texture RGBA16_SFLOAT
    Target: imported texture BGRA8_SRGB
